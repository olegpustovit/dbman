@MainActor @Observable final class QueryRunner {
  var phase: Phase = .idle     // .idle .waiting .streaming .done .failed
  func run() async {
    let slow = Task { try await Task.sleep(for: .ms(400)); phase = .waiting }
    for try await page in session.run(sql) {
      slow.cancel(); grid.append(page)   // NSTableView.insertRows
    }
    withAnimation(Motion.enter) { phase = .done }
  }
}

enum Motion {
  static var reduce: Bool {
    NSWorkspace.shared.accessibilityDisplayShouldReduceMotion }
  static var snap:  Animation { reduce ? .smooth(duration: 0.12) : .snappy(duration: 0.18) }
  static var glide: Animation { reduce ? .smooth(duration: 0.12) : .smooth(duration: 0.22) }
  static var enter: Animation { reduce ? .smooth(duration: 0.12) : .snappy(duration: 0.20) }
}
extension AnyTransition {
  static var rise: AnyTransition {
    Motion.reduce ? .opacity : .offset(y: 6).combined(with: .opacity) }
}
halftoneone Canvas pass, colours bucketed
struct Halftone: View, Equatable {
  let cells: [UInt8]; let cols: Int; let ramp: [Color]
  var body: some View {
    Canvas(rendersAsynchronously: true) { ctx, size in
      var paths = Array(repeating: Path(), count: ramp.count)
      for (i, level) in cells.enumerated() where level > 0 {
        paths[Int(level) - 1].addEllipse(in: rect(i, size))
      }
      for (p, c) in zip(paths, ramp) { ctx.fill(p, with: .color(c)) }
    }
  }
}
